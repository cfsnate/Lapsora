    """Add periodic recording segment scanner job (every 30 seconds, per D-10)."""

Create `backend/app/routers/recording.py` following the existing router pattern (see profiles.py, streams.py):

```python
"""Recording status and control endpoints."""

import logging

from fastapi import APIRouter, HTTPException

logger = logging.getLogger(__name__)

router = APIRouter(prefix="/api/recording", tags=["recording"])


@router.get("/status")
def get_recording_statuses():
    """Return recording status for all active recording processes."""
    from app.services.recording import recording_manager
    return recording_manager.get_all_statuses()


@router.get("/status/{profile_id}")
def get_recording_status(profile_id: int):
    """Return recording status for a specific profile."""
    from app.services.recording import recording_manager
    return recording_manager.get_status(profile_id)
```

This router:
- Uses `prefix="/api/recording"` following the project convention of `/api` prefix on routers
- `GET /api/recording/status` returns `dict[int, dict]` — all recording process statuses
- `GET /api/recording/status/{profile_id}` returns a single profile's status (always returns a result, with "stopped" state for profiles not actively recording)
- Imports recording_manager lazily inside route handlers (same pattern as scheduler imports in profiles.py) to avoid circular imports
- No DB dependency needed — status comes from in-memory RecordingManager state
